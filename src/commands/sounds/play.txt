require('dotenv').config();
const { SlashCommandBuilder } = require('discord.js');
const {
    joinVoiceChannel,
    createAudioPlayer,
    createAudioResource,
    AudioPlayerStatus,
    getVoiceConnection,
} = require('@discordjs/voice');
const SpotifyAPI = require('../../apis/spotifyAPI'); // Certifique-se de ajustar o caminho do arquivo SpotifyAPI

const spotify = new SpotifyAPI();

module.exports = {
    data: new SlashCommandBuilder()
        .setName('play')
        .setDescription('Toca uma prévia de uma música do Spotify.')
        .addStringOption((option) => option
            .setName('url')
            .setDescription('URL da música no Spotify')
            .setRequired(true)),

    async execute(interaction) {
        try {
            const spotifyURL = interaction.options.getString('url');
            const voiceChannel = interaction.member.voice.channel;

            if (!voiceChannel) {
                return interaction.reply({
                    content: 'Você precisa estar em um canal de voz para usar este comando!',
                    ephemeral: true,
                });
            }

            // Buscar informações da música usando a API do Spotify
            const trackInfo = await spotify.getTrack(spotifyURL);
            const previewUrl = trackInfo.preview_url;

            if (!previewUrl) {
                return interaction.reply({
                    content: 'Essa música não possui uma prévia disponível.',
                    ephemeral: true,
                });
            }

            // Conectando ao canal de voz e configurando o player
            const connection = joinVoiceChannel({
                channelId: voiceChannel.id,
                guildId: interaction.guild.id,
                adapterCreator: interaction.guild.voiceAdapterCreator,
            });

            const player = createAudioPlayer();

            player.on(AudioPlayerStatus.Idle, () => {
                console.log('Música finalizada. Saindo do canal...');
                connection.destroy();
            });

            player.on('error', (error) => {
                console.error('Erro ao reproduzir o áudio:', error);
                interaction.followUp({
                    content: 'Houve um erro ao tentar reproduzir o áudio.',
                    ephemeral: true,
                });
                connection.destroy();
            });

            const resource = createAudioResource(previewUrl);

            connection.subscribe(player);
            player.play(resource);

            // Envia a resposta ao usuário
            await interaction.reply({
                content: `🎵 Tocando: **${trackInfo.name}** por **${trackInfo.artists
                    .map((artist) => artist.name)
                    .join(', ')}** no canal: **${voiceChannel.name}**`,
            });

            // Certifique-se de que o bot não saia antes do áudio terminar
            player.once(AudioPlayerStatus.Idle, () => {
                // eslint-disable-next-line no-shadow
                const connection = getVoiceConnection(interaction.guild.id);
                if (connection) {
                    connection.destroy();
                }
            });
        } catch (error) {
            console.error('Erro no comando:', error);
            await interaction.reply({
                content: 'Houve um erro ao tentar executar o comando. Verifique a URL e tente novamente.',
                ephemeral: true,
            });
        }
    },
};
